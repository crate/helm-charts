1. CrateDB has been deployed successfully!

2. To access the CrateDB Admin UI:
   kubectl port-forward svc/{{ .Values.service.name }} 4200:4200
   
   Then navigate to: http://localhost:4200

3. To connect to CrateDB via PostgreSQL protocol:
   kubectl port-forward svc/{{ .Values.service.name }} 5432:5432
   
   Connection string: postgresql://crate@localhost:5432/

{{- if .Values.monitoring.enabled }}

   {{- if .Values.prometheus.enabled }}

   {{- end }}
   {{- if .Values.grafana.enabled }}
   - Grafana: kubectl port-forward svc/{{ include "crate.fullname" . }}-grafana 3000:80
     Default credentials: admin/{{ .Values.grafana.adminPassword }}
   {{- end }}

{{- end }}
